  function computeLineupAtMinute(initialLineup, events, minute) {
    var result = Object.assign({}, initialLineup);
    if (!events || events.length === 0) return result;
    var ordered = events.slice().sort(function(a, b) { if (a.min !== b.min) return a.min - b.min; return (a.seq || 0) - (b.seq || 0); });
    ordered.forEach(function(ev) {
      if (ev.min > minute) return;
      if (ev.type === "swap") {
        var a = result[ev.fromSlot], b = result[ev.toSlot];
        if (b) result[ev.fromSlot] = b; else delete result[ev.fromSlot];
        if (a) result[ev.toSlot] = a; else delete result[ev.toSlot];
      } else {
        Object.keys(result).forEach(function(s) { if (result[s] === ev.outId) result[s] = ev.inId; });
      }
    });
    return result;
  }
